   211→   211→
   212→   212→function extractPageCount(features: string[]): number | null {
   213→   213→  for (const f of features) {
   214→   214→    const m = f.match(/(\d+)\s*pages?/i);
   215→   215→    if (m) return parseInt(m[1], 10);
   216→   216→  }
   217→   217→  for (const f of features) {
   218→   218→    if (/one-page|single-page|1-page/i.test(f)) return 1;
   219→   219→  }
   220→   220→  return null;
   221→   221→}
