    73→/* ── Focus trap utilities ─────────────────────────────────────────── */
    74→
    75→const FOCUSABLE =
    76→  'a[href], button:not([disabled]), textarea, input, select, [tabindex]:not([tabindex="-1"])';
    77→
    78→function useFocusTrap(active: boolean, onClose: () => void) {
    79→  const ref = useRef<HTMLDivElement>(null);
    80→  const returnFocusRef = useRef<HTMLElement | null>(null);
    81→
    82→  useEffect(() => {
    83→    if (!active) return;
    84→    returnFocusRef.current = document.activeElement as HTMLElement;
    85→    document.body.style.overflow = "hidden";
    86→
    87→    const first = ref.current?.querySelector<HTMLElement>(FOCUSABLE);
    88→    requestAnimationFrame(() => first?.focus());
    89→
    90→    const onKeyDown = (e: KeyboardEvent) => {
    91→      if (e.key === "Escape") {
    92→        e.preventDefault();
    93→        onClose();
    94→        return;
    95→      }
    96→      if (e.key !== "Tab" || !ref.current) return;
    97→      const nodes = Array.from(ref.current.querySelectorAll<HTMLElement>(FOCUSABLE)).filter(
    98→        (n) => n.offsetParent !== null
    99→      );
   100→      if (nodes.length === 0) return;
   101→      const firstNode = nodes[0];
   102→      const lastNode = nodes[nodes.length - 1];
   103→      if (e.shiftKey && document.activeElement === firstNode) {
   104→        e.preventDefault();
   105→        lastNode.focus();
   106→      } else if (!e.shiftKey && document.activeElement === lastNode) {
   107→        e.preventDefault();
   108→        firstNode.focus();
   109→      }
   110→    };
   111→
   112→    document.addEventListener("keydown", onKeyDown);
   113→    return () => {
   114→      document.removeEventListener("keydown", onKeyDown);
   115→      document.body.style.overflow = "";
   116→      returnFocusRef.current?.focus?.();
   117→    };
   118→  }, [active, onClose]);
   119→
   120→  return ref;
   121→}
