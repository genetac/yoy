   362→function extractPageCount(features: string[]): number | null {
   363→  for (const f of features) {
   364→    const m = f.match(/(\d+)\s*pages?/i);
   365→    if (m) return parseInt(m[1], 10);
   366→  }
   367→  for (const f of features) {
   368→    if (/one-page|single-page|1-page/i.test(f)) return 1;
   369→  }
   370→  return null;
   371→}

   482→// ─── FamilyView: all products in a family, grouped by tier ────────
   483→
   484→function FamilyView({ familyId, onPreview }: { familyId: DesignFamilyId; onPreview: PreviewFn }) {
   485→  const fm = familyMeta[familyId];
   486→  const tokens = familyTokens[familyId];
   487→  const items = familyProductMap[familyId];
   488→  const exampleCount = items.reduce((n, fp) => n + fp.product.examples.length, 0);
   489→
   490→  // Group by tier letter for sub-sections
   491→  const byTier = useMemo(() => {
   492→    const groups: Record<string, FamilyProduct[]> = {};
   493→    for (const fp of items) {
   494→      const key = fp.tierLetter;
   495→      if (!groups[key]) groups[key] = [];
   496→      groups[key].push(fp);
   497→    }
   498→    return Object.entries(groups).sort(([a], [b]) => a.localeCompare(b));
   499→  }, [items]);
   500→
   501→  return (
   502→    <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 space-y-6">
   503→      {/* Family header */}
   504→      <div className={`rounded-lg ${fm.bg} border ${fm.border} p-4 sm:p-5 animate-slide-up`}>
   505→        <div className="flex items-start justify-between flex-wrap gap-3 mb-2">
   506→          <div className="flex items-center gap-3">
   507→            <fm.icon className={`w-5 h-5 ${fm.text}`} />
   508→            <h2 className="font-display text-xl sm:text-2xl font-bold text-zinc-900 dark:text-zinc-50">{fm.label}</h2>
   509→          </div>
   510→          <span className={`font-mono text-[10px] font-semibold px-2 py-1 rounded ${fm.chip}`}>
   511→            {items.length} products · {exampleCount} examples
   512→          </span>
   513→        </div>
   514→        <p className="text-sm text-zinc-600 dark:text-zinc-400 leading-relaxed max-w-2xl mb-3">{fm.blurb}</p>
   515→        <div className="flex items-center gap-3 font-mono text-[10px] text-zinc-500 dark:text-zinc-400">
   516→          <span>Recipe: {fm.recipe}</span>
   517→          <span>·</span>
   518→          <span>{tokens.radius.md}px radius</span>
   519→          <span>·</span>
   520→          <span>{tokens.sectionStyle}</span>
   521→        </div>
   522→      </div>
   523→
   524→      {/* Products grouped by tier */}
   525→      {byTier.map(([tierLetter, tierProducts]) => {
   526→        const tc = tierColorMap[tierProducts[0].tierColor] ?? tierColorMap.amber;
   527→        const tierName = tierProducts[0].tierName;
   528→        const priceRange = tierProducts[0].priceRange;
   529→        return (
   530→          <div key={tierLetter} className="space-y-2.5">
   531→            <div className="flex items-center gap-2 mb-1">
   532→              <span className={`flex items-center justify-center w-6 h-6 rounded font-mono text-[10px] font-bold ${tc.chip}`}>
   533→                {tierLetter}
   534→              </span>
   535→              <h3 className="font-display text-base font-bold text-zinc-900 dark:text-zinc-50">{tierName}</h3>
   536→              <span className="font-mono text-[11px] text-zinc-400 dark:text-zinc-500">{priceRange}</span>
   537→              <span className="font-mono text-[10px] text-zinc-400 dark:text-zinc-500">· {tierProducts.length} products</span>
   538→            </div>
   539→            <div className="space-y-2.5">
   540→              {tierProducts.map((fp, i) => (
   541→                <ProductSection key={fp.product.id} fp={fp} onPreview={onPreview} index={i} />
   542→              ))}
   543→            </div>
   544→          </div>
   545→        );
   546→      })}
   547→    </main>
   548→  );
   549→}
