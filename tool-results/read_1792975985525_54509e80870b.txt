   222→   222→function hexToRgb(hex) {
   223→   223→  var m = /^#?([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(String(hex).trim());
   224→   224→  if (!m) return null;
   225→   225→  var s = m[1];
   226→   226→  if (s.length === 3) s = s.charAt(0) + s.charAt(0) + s.charAt(1) + s.charAt(1) + s.charAt(2) + s.charAt(2);
   227→   227→  return [parseInt(s.slice(0, 2), 16), parseInt(s.slice(2, 4), 16), parseInt(s.slice(4, 6), 16)];
   228→   228→}
   229→   229→function normalizeHex(hex) {
   230→   230→  var rgb = hexToRgb(hex);
   231→   231→  if (!rgb) return null;
   232→   232→  return "#" + rgb.map(function (v) { var h = v.toString(16); return h.length === 1 ? "0" + h : h; }).join("").toUpperCase();
   233→   233→}
   234→   234→function linChannel(c8) {
   235→   235→  var c = c8 / 255;
   236→   236→  return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
   237→   237→}
   238→   238→function luminance(hex) {
   239→   239→  var rgb = hexToRgb(hex);
   240→   240→  if (!rgb) return null;
   241→   241→  return 0.2126 * linChannel(rgb[0]) + 0.7152 * linChannel(rgb[1]) + 0.0722 * linChannel(rgb[2]);
   242→   242→}
   243→   243→function contrastRatio(a, b) {

   247→   247→  return (hi + 0.05) / (lo + 0.05);
   248→   248→}

   764→   764→  "03-formulas/wcag-contrast-cheatsheet.md": `# WCAG 2.2 contrast — the actual math
   765→   765→
   766→   766→Every verdict in this kit comes from two formulas. No lookup tables, no
   767→   767→"eye-balling it" — this is the whole computation.
   768→   768→
   769→   769→## Formula 1 — relative luminance
   770→   770→
   771→   771→For each color, convert each sRGB channel (0–255) to a fraction
   772→   772→c = value / 255, linearize it, then take a weighted sum:
   773→   773→
   774→   774→    L = 0.2126 · R + 0.7152 · G + 0.0722 · B
   775→   775→
   776→   776→with each channel linearized as:
   777→   777→
   778→   778→    c' = c / 12.92                 if c ≤ 0.03928
   779→   779→    c' = ((c + 0.055) / 1.055)^2.4 otherwise
   780→   780→
   781→   781→The weights are the reciprocal of the sRGB primaries' contribution to
   782→   782→perceived lightness. Pure black → L = 0; pure white → L = 1.
   783→   783→
   784→   784→## Formula 2 — contrast ratio
   785→   785→
   786→   786→    C = (L1 + 0.05) / (L2 + 0.05)
   787→   787→
   788→   788→where L1 is the lighter of the two colors. The 0.05 floor accounts for
   789→   789→display flare — it is why contrast never reaches infinity and why black
   790→   790→on white tops out at exactly 21:1 (1.05 / 0.05).
   791→   791→
   792→   792→## Worked example — #2563EB (link blue) on #FFFFFF
   793→   793→
   794→   794→1. Channels: R = 0x25 = 37, G = 0x63 = 99, B = 0xEB = 235.
   795→   795→2. Fractions: 0.145098, 0.388235, 0.921569 — all above 0.03928, so
   796→   796→   each takes the power branch.
   797→   797→3. Linearized: R' = 0.018500, G' = 0.124772, B' = 0.830770.
   798→   798→4. L = 0.2126 · 0.018500 + 0.7152 · 0.124772 + 0.0722 · 0.830770
   799→   799→     = 0.153152
   800→   800→5. White: L = 1.000000.
   801→   801→6. C = (1.000000 + 0.05) / (0.153152 + 0.05) = 1.05 / 0.203152
   802→   802→     = 5.17:1 → **passes AA for normal text** (4.5), fails AAA (7).
   803→   803→
   804→   804→## Second example — the gray that quietly fails
   805→   805→
   806→   806→#777777 on white: L = 0.184475 → C = 1.05 / 0.234475 = 4.48:1 —
   807→   807→**fails AA for normal text by 0.02**. One step darker, #767676
   808→   808→(L = 0.181164), gives 4.54:1 and passes. #767676 is the lightest
   809→   809→single gray that passes AA on white; memorize it as your floor for
   810→   810→muted body text.
   811→   811→
   812→   812→## The thresholds
   813→   813→
   814→   814→| Check | Ratio | Applies to |
   815→   815→| --- | --- | --- |
   816→   816→| AA · normal text (§1.4.3) | ≥ 4.5:1 | under 24px, or under 18.66px bold |
   817→   817→| AA · large text (§1.4.3) | ≥ 3:1 | ≥ 24px, or ≥ 18.66px at 700 |
   818→   818→| AAA · normal text (§1.4.6) | ≥ 7:1 | same as AA normal |
   819→   819→| AAA · large text (§1.4.6) | ≥ 4.5:1 | same as AA large |
   820→   820→| Non-text contrast (§1.4.11) | ≥ 3:1 | borders, icons, focus rings, chart strokes |
   821→   821→

   823→   823→non-text rule (new-ish in 2.x, clarified in 2.2) is the one teams
   824→   824→forget: a 1px gray input border must still clear 3:1 against both the
   825→   825→field background and the page behind it.

   933→   933→
