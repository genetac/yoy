    75→/* ── Focus trap utilities ─────────────────────────────────────────── */
    76→
    77→const FOCUSABLE =
    78→  'a[href], button:not([disabled]), textarea, input, select, [tabindex]:not([tabindex="-1"])';
    79→
    80→function useFocusTrap(active: boolean, onClose: () => void) {
    81→  const ref = useRef<HTMLDivElement>(null);
    82→  const returnFocusRef = useRef<HTMLElement | null>(null);
    83→
    84→  useEffect(() => {
    85→    if (!active) return;
    86→    returnFocusRef.current = document.activeElement as HTMLElement;
    87→    document.body.style.overflow = "hidden";
    88→
    89→    const first = ref.current?.querySelector<HTMLElement>(FOCUSABLE);
    90→    requestAnimationFrame(() => first?.focus());
    91→
    92→    const onKeyDown = (e: KeyboardEvent) => {
    93→      if (e.key === "Escape") {
    94→        e.preventDefault();
    95→        onClose();
    96→        return;
    97→      }
    98→      if (e.key !== "Tab" || !ref.current) return;
    99→      const nodes = Array.from(ref.current.querySelectorAll<HTMLElement>(FOCUSABLE)).filter(
   100→        (n) => n.offsetParent !== null
   101→      );
   102→      if (nodes.length === 0) return;
   103→      const firstNode = nodes[0];
   104→      const lastNode = nodes[nodes.length - 1];
   105→      if (e.shiftKey && document.activeElement === firstNode) {
   106→        e.preventDefault();
   107→        lastNode.focus();
   108→      } else if (!e.shiftKey && document.activeElement === lastNode) {
   109→        e.preventDefault();
   110→        firstNode.focus();
   111→      }
   112→    };
   113→
   114→    document.addEventListener("keydown", onKeyDown);
   115→    return () => {
   116→      document.removeEventListener("keydown", onKeyDown);
   117→      document.body.style.overflow = "";
   118→      returnFocusRef.current?.focus?.();
   119→    };
   120→  }, [active, onClose]);
   121→
   122→  return ref;
   123→}
