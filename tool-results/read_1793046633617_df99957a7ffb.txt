   236→/* ─── FamilyView: all products in a family, grouped by tier ────── */
   237→
   238→function FamilyView({ familyId, onOpen }: { familyId: DesignFamilyId; onOpen: (example: ProductExample, productName: string) => void }) {
   239→  const fm = familyMeta[familyId];
   240→  const tokens = familyTokens[familyId];
   241→  const items = familyProductMap[familyId];
   242→  const exampleCount = items.reduce((n, fp) => n + fp.product.examples.length, 0);
   243→
   244→  const byTier = useMemo(() => {
   245→    const groups: Record<string, FamilyProduct[]> = {};
   246→    for (const fp of items) {
   247→      (groups[fp.tierLetter] ??= []).push(fp);
   248→    }
   249→    return Object.entries(groups).sort(([a], [b]) => a.localeCompare(b));
   250→  }, [items]);
   251→
   252→  return (
   253→    <main className="mx-auto max-w-7xl space-y-5 px-4 py-6 sm:px-6 lg:px-8">
   254→      {/* family header — prestige plate */}
   255→      <div className={`of-plate of-slide-up ${fm.bg} ${fm.border} p-4 sm:p-5`}>
   256→        <div className="mb-2 flex flex-wrap items-start justify-between gap-3">
   257→          <div className="flex items-center gap-3">
   258→            <fm.icon className={`h-5 w-5 ${fm.text}`} />
   259→            <h2 className="font-display text-xl font-semibold tracking-tight sm:text-2xl">{fm.label}</h2>
   260→          </div>
   261→          <span className={`rounded font-mono text-[10px] font-semibold px-2 py-1 ${fm.chip}`}>
   262→            {items.length} products · {exampleCount} examples
   263→          </span>
   264→        </div>
   265→        <p className="mb-3 max-w-2xl text-[13px] leading-relaxed text-zinc-600 dark:text-zinc-400">{fm.blurb}</p>
   266→        <div className="flex flex-wrap items-center gap-3 font-mono text-[10px] text-zinc-500 dark:text-zinc-400">
   267→          <span>Recipe: {fm.recipe}</span><span>·</span>
   268→          <span>{tokens.radius.md}px radius</span><span>·</span>
   269→          <span>{tokens.sectionStyle}</span><span>·</span>
   270→          <span className="of-engrave">Obsidian + brass finish</span>
   271→        </div>
   272→      </div>
   273→
   274→      {byTier.map(([tierLetter, tierProducts]) => {
   275→        const tc = tierColorMap[tierProducts[0].tierColor] ?? tierColorMap.amber;
   276→        return (
   277→          <div key={tierLetter} className="space-y-2.5">
   278→            <div className="mb-1 flex items-center gap-2">
   279→              <span className={`flex h-6 w-6 items-center justify-center rounded font-mono text-[10px] font-bold ${tc.chip}`}>{tierLetter}</span>
   280→              <h3 className="font-display text-base font-semibold tracking-tight">{tierProducts[0].tierName}</h3>
   281→              <span className="font-mono text-[11px] text-zinc-400">{tierProducts[0].priceRange}</span>
   282→              <span className="font-mono text-[10px] text-zinc-400">· {tierProducts.length} products</span>
   283→            </div>
   284→            <div className="space-y-2.5">
   285→              {tierProducts.map((fp, i) => (
   286→                <ProductSection key={fp.product.id} fp={fp} familyShort={fm.short} onOpen={onOpen} index={i} />
   287→              ))}
   288→            </div>
   289→          </div>
   290→        );
   291→      })}
   292→    </main>
   293→  );
   294→}
