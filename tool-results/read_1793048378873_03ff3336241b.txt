   238→/* ─── FamilyView: all products in a family, grouped by tier ────── */
   239→
   240→function FamilyView({ familyId, onOpen }: { familyId: DesignFamilyId; onOpen: (example: ProductExample, productName: string) => void }) {
   241→  const fm = familyMeta[familyId];
   242→  const engine = DESIGN_STYLES[useStyle()];
   243→  const tokens = familyTokens[familyId];
   244→  const items = familyProductMap[familyId];
   245→  const exampleCount = items.reduce((n, fp) => n + fp.product.examples.length, 0);
   246→
   247→  const byTier = useMemo(() => {
   248→    const groups: Record<string, FamilyProduct[]> = {};
   249→    for (const fp of items) {
   250→      (groups[fp.tierLetter] ??= []).push(fp);
   251→    }
   252→    return Object.entries(groups).sort(([a], [b]) => a.localeCompare(b));
   253→  }, [items]);
   254→
   255→  return (
   256→    <main className="mx-auto max-w-7xl space-y-5 px-4 py-6 sm:px-6 lg:px-8">
   257→      {/* family header — prestige plate */}
   258→      <div className={`of-plate of-slide-up ${fm.bg} ${fm.border} p-4 sm:p-5`}>
   259→        <div className="mb-2 flex flex-wrap items-start justify-between gap-3">
   260→          <div className="flex items-center gap-3">
   261→            <fm.icon className={`h-5 w-5 ${fm.text}`} />
   262→            <h2 className="font-display text-xl font-semibold tracking-tight sm:text-2xl">{fm.label}</h2>
   263→          </div>
   264→          <span className={`rounded font-mono text-[10px] font-semibold px-2 py-1 ${fm.chip}`}>
   265→            {items.length} products · {exampleCount} examples
   266→          </span>
   267→        </div>
   268→        <p className="mb-3 max-w-2xl text-[13px] leading-relaxed text-zinc-600 dark:text-zinc-400">{fm.blurb}</p>
   269→        <div className="flex flex-wrap items-center gap-3 font-mono text-[10px] text-zinc-500 dark:text-zinc-400">
   270→          <span>Recipe: {fm.recipe}</span><span>·</span>
   271→          <span>{tokens.radius.md}px radius</span><span>·</span>
   272→          <span>{tokens.sectionStyle}</span><span>·</span>
   273→          <span className="of-engrave">{engine.label} engine · {engine.fonts}</span>
   274→          <span className="of-chip">{engine.short} · every example renders in all 5 engines</span>
   275→        </div>
   276→      </div>
   277→
   278→      {byTier.map(([tierLetter, tierProducts]) => {
   279→        const tc = tierColorMap[tierProducts[0].tierColor] ?? tierColorMap.amber;
   280→        return (
   281→          <div key={tierLetter} className="space-y-2.5">
   282→            <div className="mb-1 flex items-center gap-2">
   283→              <span className={`flex h-6 w-6 items-center justify-center rounded font-mono text-[10px] font-bold ${tc.chip}`}>{tierLetter}</span>
   284→              <h3 className="font-display text-base font-semibold tracking-tight">{tierProducts[0].tierName}</h3>
   285→              <span className="font-mono text-[11px] text-zinc-400">{tierProducts[0].priceRange}</span>
   286→              <span className="font-mono text-[10px] text-zinc-400">· {tierProducts.length} products</span>
   287→            </div>
   288→            <div className="space-y-2.5">
   289→              {tierProducts.map((fp, i) => (
   290→                <ProductSection key={fp.product.id} fp={fp} familyShort={fm.short} onOpen={onOpen} index={i} />
   291→              ))}
   292→            </div>
   293→          </div>
   294→        );
   295→      })}
   296→    </main>
   297→  );
   298→}
