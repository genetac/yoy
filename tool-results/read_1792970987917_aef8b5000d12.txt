   211→   211→/* ── Helpers ──────────────────────────────────────────────────────── */
   212→   212→
   213→   213→function extractPageCount(features: string[]): number | null {
   214→   214→  for (const f of features) {
   215→   215→    const m = f.match(/(\d+)\s*pages?/i);
   216→   216→    if (m) return parseInt(m[1], 10);
   217→   217→  }
   218→   218→  for (const f of features) {
   219→   219→    if (/one-page|single-page|1-page/i.test(f)) return 1;
   220→   220→  }
   221→   221→  return null;
   222→   222→}
